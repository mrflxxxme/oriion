### useAuthStore (Zustand signature)

```typescript
// frontend/src/stores/auth.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

interface User {
  id: string;
  email: string;
  email_verified: boolean;
  workspace_id: string;
}

interface AuthState {
  accessToken: string | null;
  refreshToken: string | null;
  user: User | null;
  isAuthenticated: boolean;
  hydrated: boolean;

  setSession: (access: string, refresh: string, user: User) => void;
  clearSession: () => void;
  refreshAccessToken: () => Promise<void>;
  /** Hydrate from localStorage on app mount. */
  hydrate: () => void;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set, get) => ({
      accessToken: null,
      refreshToken: null,
      user: null,
      isAuthenticated: false,
      hydrated: false,
      setSession: (access, refresh, user) => set({
        accessToken: access,
        refreshToken: refresh,
        user,
        isAuthenticated: true,
      }),
      clearSession: () => set({ accessToken: null, refreshToken: null, user: null, isAuthenticated: false }),
      refreshAccessToken: async () => { /* call POST /api/v1/auth/refresh */ },
      hydrate: () => set({ hydrated: true }),
    }),
    {
      name: 'oriion-auth',
      partialize: (state) => ({ refreshToken: state.refreshToken }),  // only refresh persisted; access in-memory
    }
  )
);
```

### TaskResultRoute (page-component signature)

```tsx
// frontend/src/routes/cells.$cellId.tasks.$taskId.tsx
import { createFileRoute } from '@tanstack/react-router';
import { useQuery } from '@tanstack/react-query';
import { useTaskStream } from '@/features/tasks/hooks';
import { TaskProgressStream } from '@/features/tasks/TaskProgressStream';
import { AppShell, Breadcrumb, Card, Tabs, Badge, Button } from '@/components/ui';

export const Route = createFileRoute('/cells/$cellId/tasks/$taskId')({
  component: TaskResultPage,
  loader: async ({ params }) => ({ taskId: params.taskId, cellId: params.cellId }),
});

function TaskResultPage() {
  const { taskId, cellId } = Route.useParams();

  // TanStack Query — task metadata (status, result, cost)
  const { data: task, isLoading, isError, refetch } = useQuery({
    queryKey: ['cells', cellId, 'tasks', taskId],
    queryFn: () => tasksApi.get(cellId, taskId),
    staleTime: 5_000,
    refetchInterval: (q) => (q.state.data?.status === 'running' ? 5_000 : false),
  });

  // SSE stream — only mounted while task running
  const streamState = useTaskStream(cellId, taskId, { enabled: task?.status === 'running' });

  if (isLoading) return <TaskResultSkeleton />;
  if (isError) return <ErrorState onRetry={refetch} />;
  // ... renders Tabs (Прогресс | Результат | Стоимость) per ui-spec
}
```

### Unit test — useAuthStore hydration

```typescript
// frontend/src/stores/auth.test.ts
import { describe, it, expect, beforeEach } from 'vitest';
import { useAuthStore } from './auth';

describe('useAuthStore', () => {
  beforeEach(() => {
    useAuthStore.setState({
      accessToken: null,
      refreshToken: null,
      user: null,
      isAuthenticated: false,
      hydrated: false,
    });
  });

  it('hydrates from localStorage refresh token only (access never persisted)', () => {
    // Arrange
    localStorage.setItem('oriion-auth', JSON.stringify({
      state: { refreshToken: 'refresh-abc-123' },
      version: 0,
    }));

    // Act
    useAuthStore.persist.rehydrate();
    useAuthStore.getState().hydrate();

    // Assert
    const state = useAuthStore.getState();
    expect(state.refreshToken).toBe('refresh-abc-123');
    expect(state.accessToken).toBeNull();          // NOT persisted per security policy
    expect(state.isAuthenticated).toBe(false);     // hydration alone не authenticates
    expect(state.hydrated).toBe(true);
  });

  it('setSession transitions to authenticated state', () => {
    useAuthStore.getState().setSession(
      'access-xyz',
      'refresh-abc',
      { id: '1', email: 'test@oriion.dev', email_verified: true, workspace_id: 'ws-1' },
    );
    const state = useAuthStore.getState();
    expect(state.isAuthenticated).toBe(true);
    expect(state.user?.email).toBe('test@oriion.dev');
  });
});
```
