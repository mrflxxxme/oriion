### Retry / circuit-breaker state machine

```python
# backend/src/llm_gateway/circuit_breaker.py
from enum import Enum
from dataclasses import dataclass, field
from datetime import datetime, timedelta

class CircuitState(str, Enum):
    CLOSED = "closed"        # normal — requests pass through
    OPEN = "open"            # tripped — fast-fail; failover engaged
    HALF_OPEN = "half_open"  # probing — single request allowed

@dataclass
class ProviderCircuit:
    """Per-provider circuit breaker state.

    State transitions:
      CLOSED → OPEN: 3 consecutive failures (5xx OR timeout >30s)
      OPEN → HALF_OPEN: after cooldown_seconds (default 60s)
      HALF_OPEN → CLOSED: single successful request
      HALF_OPEN → OPEN: failed probe — reset cooldown
    """
    provider: str
    state: CircuitState = CircuitState.CLOSED
    consecutive_failures: int = 0
    last_failure_at: datetime | None = None
    opened_at: datetime | None = None
    cooldown_seconds: int = 60

    @property
    def should_attempt(self) -> bool:
        """Whether this circuit allows the next request."""
        if self.state == CircuitState.CLOSED:
            return True
        if self.state == CircuitState.OPEN and self.opened_at:
            cooldown_expired = (datetime.now() - self.opened_at) > timedelta(seconds=self.cooldown_seconds)
            if cooldown_expired:
                self.state = CircuitState.HALF_OPEN
                return True
            return False
        return self.state == CircuitState.HALF_OPEN
```

### Unit test — provider routing с circuit breaker

```python
# backend/tests/llm_gateway/test_router_service.py
import pytest
from unittest.mock import AsyncMock
from src.llm_gateway.services.router_service import LLMRouter
from src.llm_gateway.circuit_breaker import ProviderCircuit, CircuitState

@pytest.mark.asyncio
async def test_router_skips_open_circuit_and_failovers():
    """Unit: when DeepSeek circuit OPEN, router falls through к YandexGPT."""
    # Arrange
    deepseek = AsyncMock(name="deepseek")
    yandex = AsyncMock(name="yandex")
    gigachat = AsyncMock(name="gigachat")

    deepseek_circuit = ProviderCircuit(provider="deepseek", state=CircuitState.OPEN)
    yandex_circuit = ProviderCircuit(provider="yandex", state=CircuitState.CLOSED)

    router = LLMRouter(
        providers={"deepseek": deepseek, "yandex": yandex, "gigachat": gigachat},
        circuits={"deepseek": deepseek_circuit, "yandex": yandex_circuit, "gigachat": ProviderCircuit(provider="gigachat")},
    )

    # Act
    provider, model = await router.route(
        cell_id=uuid4(),
        role_key="coordinator",
        model_hint=None,
    )

    # Assert
    assert provider.name == "yandex"
    assert model.startswith("yandexgpt")
```
