### .github/workflows/deploy-staging.yml

```yaml
name: deploy-staging
on:
  push:
    branches: [main]
jobs:
  build-and-deploy:
    runs-on: ubuntu-latest
    timeout-minutes: 10        # AC: commit → production ≤ 10 min
    environment: staging
    steps:
      - uses: actions/checkout@v4

      - name: Auth к Yandex Container Registry
        run: |
          echo "${{ secrets.YC_SA_JSON }}" | docker login \
            --username json_key --password-stdin cr.yandex

      - name: Build + push backend image
        run: |
          docker build -t cr.yandex/${{ vars.YC_CR_FOLDER }}/oriion-backend:${{ github.sha }} backend/
          docker push cr.yandex/${{ vars.YC_CR_FOLDER }}/oriion-backend:${{ github.sha }}

      - name: Build + push frontend image
        run: |
          docker build -t cr.yandex/${{ vars.YC_CR_FOLDER }}/oriion-frontend:${{ github.sha }} frontend/
          docker push cr.yandex/${{ vars.YC_CR_FOLDER }}/oriion-frontend:${{ github.sha }}

      - name: SSH deploy к staging VM
        uses: appleboy/ssh-action@v1.0.3
        with:
          host: ${{ secrets.STAGING_VM_HOST }}
          username: deploy
          key: ${{ secrets.STAGING_SSH_KEY }}
          script: |
            cd /opt/oriion
            export IMAGE_TAG=${{ github.sha }}
            docker compose -f infra/docker-compose.staging.yml pull
            docker compose -f infra/docker-compose.staging.yml up -d
            ./scripts/deploy/wait_healthy.sh 120     # max 120s healthcheck wait

      - name: Smoke tests
        run: |
          for endpoint in /healthz /api/llm/providers/status; do
            curl -fsSL "https://${{ vars.STAGING_DOMAIN }}${endpoint}" || exit 1
          done

      - name: Tag deployment in Grafana annotations
        run: |
          curl -X POST "${{ vars.GRAFANA_URL }}/api/annotations" \
            -H "Authorization: Bearer ${{ secrets.GRAFANA_API_KEY }}" \
            -H "Content-Type: application/json" \
            -d '{"text":"Deploy ${{ github.sha }}","tags":["deploy","staging"]}'
```

### Caddy/Caddyfile.staging

```caddy
{$STAGING_DOMAIN} {
    tls deploy@oriion.dev

    # API → backend
    handle_path /api/* {
        reverse_proxy backend:8000
    }

    # SSE long-poll endpoints — disable buffering
    handle /api/cells/*/tasks/*/stream {
        reverse_proxy backend:8000 {
            flush_interval -1
        }
    }

    # Frontend SPA
    handle {
        root * /srv/frontend
        try_files {path} /index.html
        file_server
    }

    # Observability — basic-auth gated
    handle /grafana/* {
        basicauth {
            {$GRAFANA_BASIC_AUTH_USER} {$GRAFANA_BASIC_AUTH_BCRYPT}
        }
        reverse_proxy grafana:3000
    }

    log {
        output file /var/log/caddy/access.log
        format json
    }
}
```

```bash
# scripts/deploy/wait_healthy.sh
#!/usr/bin/env bash
# Polls /healthz until 200 OR timeout (default 120s).
# Used by deploy-staging.yml after compose-up.
set -euo pipefail
TIMEOUT="${1:-120}"
ENDPOINT="${HEALTHCHECK_URL:-http://localhost:8000/healthz}"
DEADLINE=$(( $(date +%s) + TIMEOUT ))
while (( $(date +%s) < DEADLINE )); do
    if curl -fsS "$ENDPOINT" >/dev/null 2>&1; then
        echo "✓ Backend healthy после $(($(date +%s) - DEADLINE + TIMEOUT))s"
        exit 0
    fi
    sleep 2
done
echo "✗ Healthcheck timeout (${TIMEOUT}s)" >&2
exit 1

# scripts/deploy/rollback.sh
#!/usr/bin/env bash
# Reverts backend + frontend images к previous tag (from /opt/oriion/.last-known-good).
# AC: rollback ≤ 2 min.
set -euo pipefail
PREV_TAG="$(cat /opt/oriion/.last-known-good)"
export IMAGE_TAG="${PREV_TAG}"
docker compose -f /opt/oriion/infra/docker-compose.staging.yml pull
docker compose -f /opt/oriion/infra/docker-compose.staging.yml up -d
./wait_healthy.sh 120
echo "✓ Rolled back к ${PREV_TAG}"
```

### Integration test — staging compose-up + /healthz reachable

```python
# backend/tests/_shared/observability/test_staging_smoke.py
import pytest
import requests
import time
import subprocess

@pytest.mark.integration
@pytest.mark.slow
def test_staging_compose_healthcheck():
    """Integration: bring up staging stack locally → all healthchecks green + /healthz reachable.

    Run only в CI с tags=staging-smoke. Requires Docker daemon.
    """
    subprocess.run(
        ["docker", "compose", "-f", "infra/docker-compose.staging.yml", "up", "-d"],
        check=True,
        env={
            **os.environ,
            "IMAGE_TAG": "smoke-test",
            "LOCKBOX_DATABASE_URL": "postgresql+asyncpg://oriion:oriion-test@postgres:5432/oriion_test",
            # ... other env injected from CI secrets для smoke testing
        },
    )

    try:
        # Wait для backend healthcheck (max 120s)
        deadline = time.monotonic() + 120
        while time.monotonic() < deadline:
            try:
                resp = requests.get("http://localhost:8000/healthz", timeout=2)
                if resp.status_code == 200:
                    break
            except requests.exceptions.RequestException:
                time.sleep(2)
        else:
            pytest.fail("Backend /healthz не became reachable в 120s")

        # Verify metrics endpoint
        metrics = requests.get("http://localhost:8000/metrics")
        assert metrics.status_code == 200
        assert "llm_request_total" in metrics.text

        # Verify Grafana up
        grafana = requests.get("http://localhost:3000/api/health")
        assert grafana.status_code == 200

    finally:
        subprocess.run(["docker", "compose", "-f", "infra/docker-compose.staging.yml", "down", "-v"], check=False)
```
